Program searshline ;
const
      N: = 50;
                              Var
    I, key: integer ;
    found : Boolean ;
    Keys : array [1..n] of  integer ;
                            Begin
   for i:=1 to n do
       readln(keys[I]) ;
   I : = 0 ;
   found : = false;
   readln (key);
   repeat
     I : = I+1;
     If keys [I] = key then
          found : = true;
   Until  I > n or found =true
   If  I>n then
      write (`key not found’)
   Else
      write (`key found in’ ,I);
End.
