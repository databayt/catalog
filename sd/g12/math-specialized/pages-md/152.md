∴ س̄  =  ٣٠ +  ٣٨٨ / ١٠٠   =  ٣٠ + ٣,٨٨

=  ٣٣,٨٨

مثال (٣) :

أحسب الوسط الحسابي للبيان الاحصائي المصنف في الجدول التالي :

| الفئة | ٠– | ٥– | ١٠– | ١٥– | ٢٠– | ٢٥– | ٣٠– | ٣٥– | المجموع |
| --- | --- | --- | --- | --- | --- | --- | --- | --- | --- |
| التكرار | ٢ | ٦ | ٨ | ١٨ | ٢١ | ١٦ | ٥ | ٤ | ٨٠ |

الحل :

إذا أردنا حل هذا المثال مستخدمين الوسط الفرضي يمكن أن نختار ٢٢,٥ كوسط فرضي :

| مركز الفئة م | التكرار ك | ح = م – و | ح × ك |
| --- | --- | --- | --- |
| ٢,٥ | ٢ | –٢٠ | –٤٠ |
| ٧,٥ | ٦ | –١٥ | –٩٠ |
| ١٢,٥ | ٨ | –١٠ | –٨٠ |
| ١٧,٥ | ١٨ | –٥ | –٩٠ |
| ٢٢,٥ | ٢١ | ٠ | ٠ |
| ٢٧,٥ | ١٦ | ٥ | ٨٠ |
| ٣٢,٥ | ٥ | ١٠ | ٥٠ |
| ٣٧,٥ | ٤ | ١٥ | ٦٠ |
|  | ٨٠ |  | – ١١٠ |

∑ك = ٨٠   ∑ ح × ك = – ١١٠ ، و = ٢٢,٥

∴ س̄   =   و  +  ∑ ح × ك / ∑ ك  =  ٢٢,٥ +  ‾١١٠ / ٨٠

=  ٢٢,٥ + ( ‾١,٣٧٥ ) = ٢١,١٢٥
