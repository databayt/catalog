الحل :

$\frac{٨ \sqrt{٢}}{١ – ت} = \frac{٨ \sqrt{٢} ( ١ + ت )}{( ١ – ت ) ( ١ + ت )} = \frac{٨ \sqrt{٢} ( ١ + ت )}{٢}$

= $٤ \sqrt{٢} + ٤ \sqrt{٢} ت$

∴ ر = $\sqrt{٣٢ + ٣٢}$ = $\sqrt{٦٤}$ = ٨

أ = $٤ \sqrt{٢}$ ، ب= $٤ \sqrt{٢}$ ت ⇐ ظاهـ = ١

∴ هـ = ٤٥°

ع = ٨ ( جتا ٤٥° + ت جا ٤٥° )

∴ $ع^{\frac{١}{٣}} = ٨^{\frac{١}{٣}}$ ( $جتا \frac{٤٥° + ٣٦٠° ك}{٣} + ت جا \frac{٤٥° + ٣٦٠° ك}{٣}$ )

للمعادلة ثلاثة جذور ع₀ ، ع₁ ، ع₂ تنتج بتعويض ك = ٠ ، ١ ، ٢

ع₀ = ٢ ( جتا ١٥° + ت جا ١٥° )

ع₁ = ٢ ( $جتا \frac{٤٥° + ٣٦٠°}{٣} + ت جا \frac{٤٥° + ٣٦٠°}{٣}$ )

= ٢ ( جتا ١٣٥° + ت جا ١٣٥° )

ع₂ = ٢ ( $جتا \frac{٤٥° + ٧٢٠°}{٣} + ت جا \frac{٤٥° + ٧٢٠°}{٣}$ )

= ٢ ( جتا ٢٥٥° + ت جا ٢٥٥° )

∴ جذور المعادلة هي :

٢ ( جتا ١٥° + ت جا ١٥° ) ، ٢ ( جتا ١٣٥° + ت جا ١٣٥° ) ،
٢ ( جتا ٢٥٥° + ت جا ٢٥٥° )
