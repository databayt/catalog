المانومتر الموضح بالشكل (٤–٣٧) يقيس فرق الضغط بين نقطتين "أ"، "ب" في ماء وزنه النوعي ١٠٠٠٠ نيوتن/م³ ، المانومتر يحتوي على زئبق وزنه النوعي ١٣٦٠٠٠ نيوتن/م³ احسب الفرق في الضغط بين النقطتين "أ" و "ب" إذا كان :

$ر_١$ = ٥٠ سم ، $ر_٢$ = ٤٠ سم ، $ر_٣$ = ١٥٠ سم.

افرض :

$ث_م$ = كثافة الماء، $و_م$ = الوزن النوعي للماء.

$ث_ز$ = كثافة الزئبق، $و_ز$ = الوزن النوعي للزئبق.

نرسم خط تساوي الضغط ج د،

الضغط عند "ج" = الضغط عند "د".

ض$_أ$ + $ث_م$ . د . $ر_٣$ = ض$_ب$ + $ث_م$ . د . $ر_٢$ + $ث_ز$ . د . $ر_١$

ض$_أ$ + $و_م$ . $ر_٣$ = ض$_ب$ + $و_م$ . $ر_٢$ + $و_ز$ . $ر_١$

ض$_أ$ + ١٠٠٠٠ X ١,٥ = ض$_ب$ + ١٠٠٠٠ X ٠,٤ + ١٣٦٠٠٠ X ٠,٥

ض$_أ$ + ١٥٠٠٠ = ض$_ب$ + ٤٠٠٠ + ٦٨٠٠٠

ض$_أ$ + ١٥٠٠٠ = ض$_ب$ + ٧٢٠٠٠

ض$_أ$ – ض$_ب$ = ٥٧٠٠٠ نيوتن/م²

وهو فرق الضغط بين "أ" و "ب" .
