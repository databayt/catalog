**الحل :**
حلل جميع القوى المائلة إلى مركباتها الرأسية والأفقية لتحصل على مجموعة من القوى في اتجاهين فقط كما موضح بالشكل( ٤–٩ ) أدناه .

**الشكل ( ٤–٩ ) :تحليل جميع القوى الى مركبات رأسية و أفقية .**

- ٣٠ جا ٤٥°
- ٤٠ جتا ٣٠°
- ٥٠ جا ٤٥°
- ٣٠ جتا ٤٥°
- ٦٠ جتا ٣٠°
- ٥٠ جتا ٤٥°
- ٤٠ جا ٣٠°
- ٢٠ جا ٦٠°
- ١٥
- ٦٠ جا ٣٠°
- ٢٠ جتا ٦٠°
- ٢٥

نجمع القوى الأفقية للحصول على( ق س ) :

ق س = ١٥ + ٢٠جا٦٠ + ٥٠جتا٤٥ + ٤٠جا٣٠ – ٦٠جتا٣٠ – ٣٠جتا٤٥

ق س = ١٥ + ٢٠ X ٠,٩ + ٥٠ X ٠,٧ + ٤٠ X ٠,٥ – ٦٠ X ٠,٩ – ٣٠ X ٠,٧

ق س = ١٥ + ١٨ + ٣٥ + ٢٠ – ٥٤ – ٢١ = ١٣ نيوتن

نجمع القوى الراسية للحصول على( ق ص ) :

ق ص = ٣٠جا٤٥ + ٤٠جتا٣٠ + ٥٠جا٤٥ – ٦٠جا٣٠ – ٢٠جتا٦٠ – ٢٥

ق ص = ٣٠ X ٠,٧ + ٤٠ X ٠,٩ + ٥٠ X ٠,٧ – ٦٠ X ٠,٥ – ٢٠ X ٠,٥ – ٢٥

ق ص = ٢١ + ٣٦ + ٣٥ – ٣٠ – ١٠ – ٢٥ = ٢٧ نيوتن

نحسب المحصلة " ح " :

$ح^٢ = ق_س^٢ + ق_ص^٢$

$ح^٢ = ١٣^٢ + ٢٧^٢ = ١٦٩ + ٧٢٩ = ٨٩٨$

ح = ٣٠ نيوتن
